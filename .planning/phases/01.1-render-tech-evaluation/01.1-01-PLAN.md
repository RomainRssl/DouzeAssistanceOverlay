Then rewrite UpdateAll() to use the cached pattern from VROverlayService.CaptureAndSubmit() instead of allocating per frame:

```csharp
public void UpdateAll()
{
    if (!IsVRActive) return;
    foreach (var ovl in _ovls.Values)
    {
        if (!ovl.Settings.IsEnabled) continue;
        var content = ovl.Window.Content as FrameworkElement;
        if (content == null) continue;
        double scale = Math.Max(1.0, ovl.Settings.Scale);
        int w = Math.Max(1, (int)(content.ActualWidth  * scale));
        int h = Math.Max(1, (int)(content.ActualHeight * scale));
        if (w < 1 || h < 1) continue;

        // Cache RTB — only reallocate when dimensions change
        if (ovl.CachedRtb == null || ovl.LastPixelW != w || ovl.LastPixelH != h)
        {
            ovl.CachedRtb    = new RenderTargetBitmap(w, h, 96 * scale, 96 * scale,
                                   System.Windows.Media.PixelFormats.Pbgra32);
            ovl.CachedPixels = new byte[h * w * 4];
            ovl.LastPixelW   = w;
            ovl.LastPixelH   = h;
        }

        ovl.CachedRtb.Render(content);
        int stride = w * 4;
        ovl.CachedRtb.CopyPixels(ovl.CachedPixels!, stride, 0);
        byte[] pixels = ovl.CachedPixels!;

        // BGRA → RGBA
        for (int i = 0; i < pixels.Length; i += 4)
        {
            byte b = pixels[i];
            pixels[i]     = pixels[i + 2];
            pixels[i + 2] = b;
        }

        lock (ovl.Lock) { ovl.PixelBuf = pixels; ovl.Dirty = true; }
    }
}
```

DO NOT change the pixel buffer locking logic or the PixelBuf assignment pattern — only the allocation strategy.
DO NOT touch the rest of OpenXRService (FrameLoop, UploadPixels, SetOverlayPosition, etc.).

public override void UpdateData()
{
    long t0 = System.Diagnostics.Stopwatch.GetTimestamp();

    // --- existing render body (unchanged) ---
    foreach (var el in _dynamicElements)
        _radarCanvas.Children.Remove(el);
    _dynamicElements.Clear();
    // ... rest of existing code unchanged ...

    // --- end of render body ---
    long t1 = System.Diagnostics.Stopwatch.GetTimestamp();
    long us = (t1 - t0) / _ticksPerUs;
    _renderTimes.Enqueue(us);
    if (_renderTimes.Count > 500) _renderTimes.Dequeue();

    _sampleCount++;
    if (_sampleCount >= 500)
    {
        _sampleCount = 0;
        var arr = _renderTimes.ToArray();
        Array.Sort(arr);
        long min  = arr[0];
        long avg  = (long)arr.Average();
        long p95  = arr[(int)(arr.Length * 0.95)];
        long max  = arr[arr.Length - 1];
        System.Diagnostics.Debug.WriteLine(
            $"[ProximityRadar WPF baseline] min={min}us avg={avg}us p95={p95}us max={max}us (500 samples)");
    }
}
