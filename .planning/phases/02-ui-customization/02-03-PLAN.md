    /// <summary>
    /// Deploys the 3 preset theme JSON files to the themes directory.
    /// Uses WritePresetIfAbsent — does NOT overwrite user-edited files.
    /// Called at startup in App.xaml.cs after EnsureDefaultThemeExists().
    /// </summary>
    public static void EnsurePresetThemesExist()
        => EnsurePresetThemesExistIn(ThemesDirectory);

    /// <summary>
    /// Overload accepting explicit directory — used by unit tests with a temp dir.
    /// </summary>
    public static void EnsurePresetThemesExistIn(string directory)
    {
        Directory.CreateDirectory(directory);
        WritePresetIfAbsent(Path.Combine(directory, "racing-clair.json"),       BuildRacingClair());
        WritePresetIfAbsent(Path.Combine(directory, "bleu-nuit-lmp2.json"),     BuildBleuNuitLmp2());
        WritePresetIfAbsent(Path.Combine(directory, "minimaliste-mono.json"),   BuildMinimalisteMono());
    }

    private static void WritePresetIfAbsent(string path, string json)
    {
        if (!File.Exists(path))
            File.WriteAllText(path, json, System.Text.Encoding.UTF8);
    }
